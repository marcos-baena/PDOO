def self.discard_element(uses_left)
  if uses_left == @@MAX_USES
    result=false;
  elsif uses_left == 0
    result=true;
  else
    result= @generator.rand < (@@MAX_USES-uses_left).to_f/@@MAX_USES 
  end                                                 
  result
end
